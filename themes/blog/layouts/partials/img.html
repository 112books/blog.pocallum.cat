{{/* ── Imatge responsiva variant = "" (cap webp) | "hero" (-2500) | "thumb" (-800) Per a imatges del repo (/media/): només si el WebP existeix a static/ (les pujades noves del CMS poden no tenir-lo — fallback a ). Per a imatges del llegat (/wp-content/): si imatgesWebp = true. ── */}} {{- $src := .src | default "" -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $decoding := .decoding | default "async" -}} {{- $variant := .variant | default "" -}} {{- $webpActivat := site.Params.imatgeswebp | default false -}} {{- $dim := index site.Data.imatges_dim $src -}} {{- $esRepo := or (strings.HasPrefix $src "/media/") (strings.HasPrefix $src "https://blog.pocallum.cat/media/") -}} {{- $esWp := strings.HasPrefix $src "https://blog.pocallum.cat/wp-content/" -}} {{- $w800 := "" -}} {{- $w2500 := "" -}} {{- $re := `\.(jpg|jpeg|png)(\?.*)?$` -}} {{- if and $variant $src (findRE $re $src) -}} {{- $w800 = replaceRE $re "-800.webp" $src -}} {{- $w2500 = replaceRE $re "-2500.webp" $src -}} {{- end -}} {{- $doWebp := false -}} {{- if $w800 -}} {{- if $esRepo -}} {{- $doWebp = os.FileExists (printf "static%s" $w800) -}} {{- else if and $esWp $webpActivat -}} {{- $doWebp = true -}} {{- end -}} {{- end -}} {{- $w800w := 800 -}} {{- $w2500w := 2500 -}} {{- with $dim -}} {{- $ow := index . 0 -}} {{- $w800w = cond (gt $ow 800) 800 $ow -}} {{- $w2500w = cond (gt $ow 2500) 2500 $ow -}} {{- end -}} {{- $sizes := .sizes | default "" -}} {{- if not $sizes -}} {{- if eq $variant "hero" -}}{{- $sizes = "100vw" -}}{{- else -}}{{- $sizes = "(max-width: 540px) 50vw, (max-width: 900px) 33vw, 220px" -}}{{- end -}} {{- end -}} {{- if $doWebp -}} {{- if and $dim (gt $w2500w $w800w) }} {{- else }} {{- end }} {{- else -}} {{- end -}}
